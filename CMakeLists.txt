cmake_minimum_required(VERSION 3.16)
project(CreatureLab VERSION 0.12.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(creature SHARED src/sim.cpp src/content.cpp src/api.cpp)
target_include_directories(creature PUBLIC include)
add_executable(sim_tests src/sim.cpp src/content.cpp tests/sim_tests.cpp)
target_include_directories(sim_tests PRIVATE include)
add_executable(benchmark src/sim.cpp src/content.cpp tests/benchmark.cpp)
target_include_directories(benchmark PRIVATE include)
enable_testing()
add_test(NAME simulation COMMAND sim_tests)
option(CREATURE_VIEWER "Build SDL2 viewer" ON)
if(CREATURE_VIEWER)
  find_package(SDL2 CONFIG QUIET)
  if(NOT TARGET SDL2::SDL2)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
    add_library(SDL2::SDL2 ALIAS PkgConfig::SDL2)
  endif()
  add_executable(creature_lab src/sim.cpp src/content.cpp agents/brain.cpp src/campaign.cpp src/campaign_content.cpp client/main.cpp)
  target_include_directories(creature_lab PRIVATE include)
  target_link_libraries(creature_lab PRIVATE SDL2::SDL2)
  add_executable(journey_audio_tests tests/journey_audio_tests.cpp)
  target_link_libraries(journey_audio_tests PRIVATE SDL2::SDL2)
  add_test(NAME journey_audio COMMAND journey_audio_tests)
endif()
add_executable(tournament src/sim.cpp src/content.cpp tests/tournament.cpp)
target_include_directories(tournament PRIVATE include)
add_executable(counterplay src/sim.cpp src/content.cpp tests/counterplay.cpp)
target_include_directories(counterplay PRIVATE include)

add_executable(environment_tests src/sim.cpp src/content.cpp tests/environment_tests.cpp)

add_executable(brain_tests src/sim.cpp src/content.cpp agents/brain.cpp tests/brain_tests.cpp)
target_include_directories(brain_tests PRIVATE include)

add_executable(brain_eval src/sim.cpp src/content.cpp agents/brain.cpp tests/brain_eval.cpp)

add_executable(brain_probe src/sim.cpp src/content.cpp agents/brain.cpp tests/brain_probe.cpp)

target_include_directories(campaign_playthrough PRIVATE include)
add_executable(footwork_benchmark src/sim.cpp src/content.cpp tests/footwork_benchmark.cpp)
